cmake_minimum_required(VERSION 3.21)
project(companion_native_admission LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Test)
add_library(admissiongate STATIC admissiongate.cpp admissiongate.h)
target_link_libraries(admissiongate PUBLIC Qt6::Core Qt6::Gui)
target_include_directories(admissiongate PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(admissiongate PRIVATE -Wall -Wextra -Werror)
add_library(companionadmissionplugin MODULE plugin.cpp)
target_link_libraries(companionadmissionplugin PRIVATE admissiongate Qt6::Qml)
set_target_properties(companionadmissionplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qml/Companion/Admission")
configure_file(qmldir "${CMAKE_CURRENT_BINARY_DIR}/qml/Companion/Admission/qmldir" COPYONLY)
include(CTest)
if(BUILD_TESTING)
    add_executable(tst_admissiongate ../tests/native-admission/tst_admissiongate.cpp)
    target_link_libraries(tst_admissiongate PRIVATE admissiongate Qt6::Test)
    target_compile_options(tst_admissiongate PRIVATE -Wall -Wextra -Werror)
    add_test(NAME admissiongate COMMAND tst_admissiongate)
    set_tests_properties(admissiongate PROPERTIES TIMEOUT 20)
endif()
